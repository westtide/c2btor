cmake_minimum_required(VERSION 3.16)
project(C2BTOR LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_BINARY_DIR)
  message(FATAL_ERROR "Use an out-of-source build: cmake -S . -B build")
endif()
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()

file(STRINGS src/config.inc C2BTOR_VERSION REGEX "^C2BTOR_VERSION = ")
string(REGEX REPLACE "C2BTOR_VERSION = " "" C2BTOR_VERSION "${C2BTOR_VERSION}")
file(STRINGS src/config.inc CBMC_VERSION REGEX "^CBMC_VERSION = ")
string(REGEX REPLACE "CBMC_VERSION = " "" CBMC_VERSION "${CBMC_VERSION}")
set(PROJECT_VERSION "${C2BTOR_VERSION}")
string(REPLACE "." ";" version_parts "${C2BTOR_VERSION}")
list(GET version_parts 0 PROJECT_VERSION_MAJOR)
list(GET version_parts 1 PROJECT_VERSION_MINOR)
list(GET version_parts 2 PROJECT_VERSION_PATCH)
message(STATUS "Building C2BTOR ${C2BTOR_VERSION} with CBMC frontend ${CBMC_VERSION}")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS src/config.inc)

# Names used by the retained upstream build modules.
set(CBMC_SOURCE_DIR ${CMAKE_CURRENT_SOURCE_DIR})
set(CBMC_BINARY_DIR ${CMAKE_CURRENT_BINARY_DIR})
set(CBMC_ROOT_DIR ${CMAKE_CURRENT_SOURCE_DIR})
include(GNUInstallDirs)
set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/lib)
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/lib)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

find_program(CCACHE_PROGRAM ccache)
if(CCACHE_PROGRAM)
  set(CMAKE_C_COMPILER_LAUNCHER "${CCACHE_PROGRAM}")
  set(CMAKE_CXX_COMPILER_LAUNCHER "${CCACHE_PROGRAM}")
endif()
if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
  set(CMAKE_CXX_FLAGS_RELEASE "-O2")
  set(CMAKE_CXX_FLAGS_RELWITHDEBINFO "-O2 -g")
  add_compile_options(-Wall -Wpedantic -Werror -Wswitch-enum
    -Wno-deprecated-declarations -Wno-maybe-uninitialized)
elseif(CMAKE_CXX_COMPILER_ID MATCHES "Clang")
  set(CMAKE_CXX_FLAGS_RELEASE "-O2")
  set(CMAKE_CXX_FLAGS_RELWITHDEBINFO "-O2 -g")
  add_compile_options(-Wall -Wpedantic -Werror -Wswitch-enum
    -Wno-deprecated-declarations)
elseif(MSVC)
  find_package(Git REQUIRED)
  get_filename_component(git_root "${GIT_EXECUTABLE}" DIRECTORY)
  set(ENV{PATH} "${git_root}/../usr/bin;$ENV{PATH}")
endif()

set(sat_impl minisat2 CACHE STRING "Internal dependency: minisat2 or system-minisat2")
set_property(CACHE sat_impl PROPERTY STRINGS minisat2 system-minisat2)
if(NOT sat_impl STREQUAL "minisat2" AND NOT sat_impl STREQUAL "system-minisat2")
  message(FATAL_ERROR "C2BTOR supports sat_impl=minisat2 or system-minisat2")
endif()
option(WITH_JBMC "Compatibility option; Java is not part of C2BTOR" OFF)
if(WITH_JBMC)
  message(FATAL_ERROR "JBMC is not included; use -DWITH_JBMC=OFF")
endif()
option(enable_cbmc_tests "Build focused C2BTOR, C frontend and GOTO-IR tests" OFF)
if(NOT MSVC)
  find_package(Perl REQUIRED)
endif()

# Only the product dependency graph participates in the default build.
# converter/file_converter are required build-time C library generators.
add_subdirectory(src)
if(enable_cbmc_tests)
  enable_testing()
  add_subdirectory(unit)
endif()
include(cmake/packaging.cmake)
